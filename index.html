SQL Supabase

Jalankan SQL ini sekali di Supabase SQL Editor sebelum upload dari tab Admin.

create table if not exists public.sales_purchase_history (
  id bigint generated by default as identity primary key,
  batch_name text,
  sales_name text,
  customer_name text not null,
  transaction_date date,
  item_name text not null,
  qty numeric default 0,
  invoice_no text,
  sales_value numeric default 0,
  raw_data jsonb,
  uploaded_at timestamptz default now()
);

create index if not exists idx_sales_purchase_history_customer
  on public.sales_purchase_history (customer_name);

create index if not exists idx_sales_purchase_history_sales
  on public.sales_purchase_history (sales_name);

create index if not exists idx_sales_purchase_history_date
  on public.sales_purchase_history (transaction_date desc);

alter table public.sales_purchase_history enable row level security;

create policy "allow read purchase history"
  on public.sales_purchase_history
  for select
  using (true);

create policy "allow anon insert purchase history"
  on public.sales_purchase_history
  for insert
  with check (true);

create policy "allow anon delete by batch"
  on public.sales_purchase_history
  for delete
  using (true);

Catatan: policy di atas dibuat praktis untuk prototype. Untuk produksi, sebaiknya admin memakai Supabase Auth, lalu policy insert/delete hanya diberikan kepada role admin.
